"""Retain observed local shadow evidence with an explicit capture gap."""
import argparse
import hashlib
import json
from pathlib import Path
import re
import sys
from urllib.parse import unquote

parser = argparse.ArgumentParser()
parser.add_argument('--repo', type=Path, required=True)
args = parser.parse_args()
repo = args.repo.resolve()
scratch = Path(__file__).resolve().parent
local = repo / '.rwb/contract-shadow'
attempt = repo / 'work/TEST-PERF-002/A-20260916-006'
attempt.mkdir(parents=True, exist_ok=True)
(attempt / 'checks').mkdir(exist_ok=False)
(attempt / '.gitattributes').write_bytes(b'* -text whitespace=cr-at-eol\n')
copies = {
    local / 'shadow-focused.log': 'shadow-focused.log',
    local / 'shadow-coverage.json': 'shadow-coverage.json',
    local / 'regression.log': 'regression.log',
    local / 'pr84.json': 'pr84-shadow.json',
    local / 'pr84-summary.json': 'pr84-shadow-summary.json',
    local / 'implementation/ci-plan.json': 'implementation-plan.json',
    local / 'implementation/preflight-summary.json': 'implementation-preflight.json',
    local / 'preflight.log': 'implementation-preflight.log',
    repo / '.github/scripts/ci_contract_shadow.py': 'ci_contract_shadow.py.txt',
    repo / 'tests/test_ci_contract_shadow.py': 'test_ci_contract_shadow.py.txt',
    scratch / 'shadow_preflight.py': 'shadow_preflight.py.txt',
    Path(__file__).resolve(): 'archive_shadow.py.txt',
}
manifest = []
for source, name in copies.items():
    raw = source.read_bytes()
    (attempt / 'checks' / name).write_bytes(raw)
    manifest.append({'path': 'checks/' + name, 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)})
(attempt / 'checks/evidence-manifest.json').write_text(json.dumps(manifest, indent=2)+'\n', encoding='utf-8')
preflight = json.loads((local / 'implementation/preflight-summary.json').read_bytes())
coverage = json.loads((local / 'shadow-coverage.json').read_bytes())['totals']
assert coverage['covered_lines'] == coverage['num_statements'] == 140
assert coverage['covered_branches'] == coverage['num_branches'] == 58
assert not coverage['excluded_lines']
assert 'Ran 189 tests' in (local / 'regression.log').read_text(encoding='utf-8')
assert (local / 'regression.log').read_text(encoding='utf-8').rstrip().endswith('OK')
(attempt / 'RESULTS.md').write_text(f'''# TEST-PERF-002 contract shadow v1 evidence

Owner: Chengyue-Lu. Risk: R2. Attempt: A-20260916-006. User authorized the first implementation, Draft PR and Issue #48 update.

## Retained local results

- Implementation commit: `{preflight['head']}`; base: `{preflight['base']}`.
- New module: 12 PASS; 140/140 statements and 58/58 branches, zero exclusions.
- Existing dependency/planner/checker/witness/coverage-policy/documentation regression: 189 PASS in 260.663 s. Together: 201 PASS; this is focused regression, not a repository full-suite result.
- Exact implementation-head plan, worker recomputation and governance: PASS. Obligations: full behavioral, impact + repository coverage. Actual impact subject: `.github/scripts/ci_contract_shadow.py` only.
- PR84 replay: 15 document, 8 evidence-data, 1 archive-attributes, 9 classification conflicts; retained dependency selection 91/92. Diagnostic analysis: 7.360 s on this local run. CI savings have not yet been established.
- Existing quality thresholds, omitted modules, coverage exclusions and selection implementations remain unchanged; the diagnostic workflow is the declared authority-sensitive change.

The implementation plan predates this archive commit. Final-head plan/governance checks and hosted CI are separate PR evidence; archived source hashes identify the tested bytes. This archive does not attest hosted full execution or human acceptance.

## Scope and navigation

[Implementation and boundaries](../../../docs/workstreams/chengyue-lu/TEST-PERF-002/SHADOW_V1.md), [source and check hashes](checks/evidence-manifest.json), [implementation preflight](checks/implementation-preflight.json), [PR84 report](checks/pr84-shadow.json), [Trace validation](checks/trace-validation.json).

The shadow report preserves all accepted obligations and cannot control skip or impersonate plan v4. Inputs/contracts still require reviewed consumer boundaries and independent witness support before activation.

Trace is retrospective retention of the observed results, with an explicit capture-gap warning. It is not a complete tool-event stream. Historical evidence and the deliberately failing document probe retain their original outcomes.
''', encoding='utf-8')
sys.path.insert(0, str(repo / 'src'))
from research_workbench.observability.trace import AgentTraceRecorder, validate_attempt_trace
recorder = AgentTraceRecorder(attempt, task_id='TEST-PERF-002', task_revision=34,
    attempt_id='A-20260916-006', task_snapshot={'task_id':'TEST-PERF-002', 'revision':34,
        'request':'我认可，尝试进行一版并提交到DraftPR，同时跟进到issue下。',
        'scope':'P1 typed input and consumer shadow report, preserved authoritative CI obligations',
        'delegation':False, 'required_skills':[], 'implementation_commit':preflight['head']},
    accountable_owner='Chengyue-Lu', actor_id='main-agent', runtime_identity='Codex desktop', provider='OpenAI',
    read_allowlist=['AGENTS.md','docs/**','src/**','tests/**','.github/**','build_backend.py',
        'runtime-resources.json','pyproject.toml','work/TEST-PERF-002/**'],
    write_scope=['.github/scripts/ci_contract_shadow.py','.github/workflows/ci.yml',
        'tests/test_ci_contract_shadow.py','tests/coverage_policy.yaml',
        'docs/workstreams/chengyue-lu/TEST-PERF-002/**','work/TEST-PERF-002/A-20260916-006/**'],
    tool_allowlist=['git','gh','python','shell','apply_patch'], baseline=preflight['base'])
recorder.record_capture_gap('events',
    'Retrospective retention of local observed results. Earlier shell and tool exchanges were not streamed into this recorder; complete event capture is not claimed.')
recorder.record_tool_call(operation_id='retain-local-shadow-evidence', tool_name='python', status='succeeded',
    arguments={'operation':'retain exact source, tests, replay and local preflight outputs'},
    result={'new_tests':12, 'existing_regression':189, 'shadow_coverage':coverage,
            'implementation_preflight':preflight, 'retained_files':manifest}, result_entered_context=True)
recorder._index['check_refs'] = [{'path':row['path'],'sha256':row['sha256']} for row in manifest]
recorder._refresh_index()
recorder.seal('safe-paused')
validation = validate_attempt_trace(repo, attempt)
result = {'blocked':validation.blocked,
    'risks':[{'code':r.code,'level':r.level.value,'message':r.message} for r in validation.risks]}
(attempt / 'checks/trace-validation.json').write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
assert not validation.blocked
links = 0
files = list((repo/'docs/workstreams/chengyue-lu/TEST-PERF-002').glob('*.md')) + [attempt/'RESULTS.md']
for file in files:
    for target in re.findall(r'\]\(([^)]+)\)', file.read_text(encoding='utf-8')):
        target = target.strip('<>').split('#',1)[0]
        if not target or '://' in target or target.startswith('mailto:'):
            continue
        assert (file.parent / unquote(target)).resolve().exists(), (file.relative_to(repo), target)
        links += 1
print(json.dumps({'trace':result, 'retained_files':len(manifest), 'local_links_checked':links}, indent=2))
