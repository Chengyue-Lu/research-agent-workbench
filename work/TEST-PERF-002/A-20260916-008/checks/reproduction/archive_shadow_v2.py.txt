import argparse
import hashlib
import json
from pathlib import Path
import re
import subprocess
import sys

p = argparse.ArgumentParser()
p.add_argument('--repo', type=Path, required=True)
a = p.parse_args()
repo = a.repo.resolve()
scratch = Path(__file__).resolve().parent
source = repo / '.rwb/contract-shadow'
attempt = repo / 'work/TEST-PERF-002/A-20260916-008'
assert sorted(x.name for x in attempt.iterdir()) == ['RESULTS.md'], 'archive already populated'
(attempt / 'checks').mkdir(exist_ok=False)
(attempt / '.gitattributes').write_bytes(b'* -text whitespace=cr-at-eol\n')
comparison = json.loads((source / 'corpus-comparison.json').read_bytes())
assert len(comparison['cases']) == 12
assert all(c['status'] == 'PASS' for c in comparison['cases'])
assert len(comparison['preserved_execution_fields']) == 16
preflight = json.loads((source / 'v2-implementation/preflight-summary.json').read_bytes())
assert preflight['head'] == 'c713f999d7e39900966f6e59156377fc0dc3ec93'
assert all(x['checkout_sha256'] == x['git_sha256'] for x in preflight['source_manifest'])
assert subprocess.check_output(['git', '-C', str(repo), 'rev-parse', 'HEAD']).decode().strip() == preflight['head']
coverage = json.loads((source / 'v2-coverage.json').read_bytes())
coverage_summary = {k: v['summary'] for k, v in coverage['files'].items()}
assert len(coverage_summary) == 3
assert all(s['missing_lines'] == 0 and s['missing_branches'] == 0 and s['excluded_lines'] == 0 for s in coverage_summary.values())
executions = []
for name, expected in [('v2-focused.log', 72), ('v2-regression.log', 137)]:
    log = (source / name).read_text(encoding='utf-8')
    match = re.search(r'^Ran (\d+) tests in ([\d.]+)s$', log, re.M)
    assert match and int(match[1]) == expected and '\nOK\n' in log
    executions.append({'log': name, 'tests': expected, 'seconds': float(match[2]), 'status': 'PASS'})
copies = {}
for dirname in ('corpus-before', 'corpus-after', 'v2-implementation'):
    for file in (source / dirname).iterdir():
        if file.suffix == '.json':
            copies[file] = dirname + '/' + file.name
for name in ('v2-focused.log', 'v2-regression.log', 'v2-coverage.json', 'corpus-comparison.json',
             'corpus-comparison.log', 'v2-preflight.log'):
    copies[source / name] = name
for file in scratch.glob('corpus-pr*.json'):
    copies[file] = 'pr-metadata/' + file.name
for name in ('replay_matrix.py', 'compare_matrix.py', 'shadow_preflight_v2.py', 'archive_shadow_v2.py'):
    copies[scratch / name] = 'reproduction/' + name + '.txt'
for name in ('ci_input_facts.py', 'ci_contract_shadow.py', 'ci_dependencies.py', 'plan_ci.py', 'check_pr_governance.py'):
    copies[repo / '.github/scripts' / name] = 'producer/' + name + '.txt'
manifest = []
for file, name in sorted(copies.items(), key=lambda item: item[1]):
    raw = file.read_bytes()
    destination = attempt / 'checks' / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(raw)
    manifest.append({'path': 'checks/' + name, 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)})
(attempt / 'checks/evidence-manifest.json').write_text(json.dumps(manifest, indent=2) + '\n', encoding='utf-8', newline='\n')
